Swift constants (D-15), and stand up the new `CortexReFIT` Swift package scaffold with the seam
decision locked (D-14). This is the foundation Wave-2's filter consumes — the Swift hot path loads

and bakes the result into a committed constants file so the Swift filter is data-grounded and
reproducible.

    ```python
    from __future__ import annotations
    import numpy as np
    from scipy import linalg as la

    DT: float = 0.020  # 20 ms tick (CONTEXT D-01)

    def full_transition() -> np.ndarray:
        # 6×6 constant-acceleration A on [px,py,vx,vy,ax,ay] (RESEARCH §2.1):
        #   [ I  dt·I  ½dt²·I ]
        #   [ 0  I     dt·I   ]
        #   [ 0  0     I      ]
        I2 = np.eye(2); Z2 = np.zeros((2, 2)); h = 0.5 * DT * DT
        return np.block([[I2, DT*I2, h*I2],
                         [Z2, I2,    DT*I2],
                         [Z2, Z2,    I2]])

    def full_measurement() -> np.ndarray:
        # 2×6 H = [0 I 0] — measures (vx,vy) only (RESEARCH §2.1)
        I2 = np.eye(2); Z2 = np.zeros((2, 2))
        return np.block([[Z2, I2, Z2]])

    def observable_blocks() -> tuple[np.ndarray, np.ndarray]:
        # 4-DOF observable sub-block on [vx,vy,ax,ay] (RESEARCH §2.3):
        #   A_obs = [ I  dt·I ]   H_obs = [ I  0 ]
        #           [ 0  I    ]
        I2 = np.eye(2); Z2 = np.zeros((2, 2))
        A_obs = np.block([[I2, DT*I2], [Z2, I2]])
        H_obs = np.block([[I2, Z2]])
        return A_obs, H_obs

    def steady_state_gain(Q_obs: np.ndarray, R: np.ndarray) -> np.ndarray:
        """Solve the steady-state Kalman gain on the OBSERVABLE block, return 6×2 K with zero position rows.
        RESEARCH §2.3/§3.1: the full 6×6 DARE has no stabilizing solution (velocity-only measurement ⇒
        position unobservable). Solve the dual on the 4-DOF block, embed K_obs into rows 2..6, zero rows 0,1."""
        A_obs, H_obs = observable_blocks()
        # scipy solves the CONTROL-form DARE; the Kalman a-priori covariance is the DUAL → pass A.T, H.T (RESEARCH §3.1)
        P_pred = la.solve_discrete_are(A_obs.T, H_obs.T, Q_obs, R)        # 4×4 a-priori covariance
        S = H_obs @ P_pred @ H_obs.T + R                                  # 2×2 innovation covariance
        K_obs = P_pred @ H_obs.T @ la.inv(S)                              # 4×2 steady-state gain
        # Schur-stability assertion: closed-loop A_obs(I − K_obs H_obs) eigenvalues |λ|<1 (RESEARCH §3.1)
        cl = A_obs @ (np.eye(4) - K_obs @ H_obs)
        eig_max = float(np.max(np.abs(np.linalg.eigvals(cl))))
        if not eig_max < 1.0:
            raise ValueError(f"closed-loop not Schur-stable: max|lambda|={eig_max:.6f} >= 1")
        K = np.zeros((6, 2)); K[2:6, :] = K_obs                           # zero position rows → 6×2
        return K
    ```
