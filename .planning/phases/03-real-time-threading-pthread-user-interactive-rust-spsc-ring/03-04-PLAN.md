```c
typedef struct CortexFrame { uint64_t ts_ns; uint64_t seq; uint16_t channel_data[96]; } CortexFrame;
typedef struct CortexSpsc CortexSpsc;
uint32_t  cortex_ping(uint32_t x);
CortexSpsc *cortex_spsc_create(size_t capacity);     // power-of-two
bool      cortex_spsc_push(CortexSpsc *r, const CortexFrame *f);  // false = full
bool      cortex_spsc_pop(CortexSpsc *r, CortexFrame *out);       // false = empty
void      cortex_spsc_destroy(CortexSpsc *r);
```

<!-- Swift wrapper shape (Sources/CortexRing/Ring.swift) — safe RAII over the C ABI. -->
```swift
import CortexRingFFI
public final class CortexRing {
    private let handle: OpaquePointer            // *mut CortexSpsc
    public init?(capacity: Int)                  // nil if create returns null
    deinit { cortex_spsc_destroy(handle) }       // RAII destroy
    public func push(_ frame: CortexFrame) -> Bool   // CortexFrame is the C type from the modulemap
    public func pop() -> CortexFrame?
}
```
